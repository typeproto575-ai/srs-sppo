// Include required files: database connection and helper functions
require_once __DIR__ . '/../config/database.php';

// Subfolder-safe URL prefix for the CTA links below.
$basePath = $basePath ?? base_path();
// Load secretary-editable public page content (site_settings table from migration 007).
$settings = [];
try {
    $tableExists = (bool) $pdo->query(
        "SELECT 1 FROM information_schema.tables " .
        "WHERE table_schema = DATABASE() AND table_name = 'site_settings'"
    )->fetch();
    if ($tableExists) {
        foreach ($pdo->query("SELECT setting_key, setting_value FROM site_settings ORDER BY setting_id")->fetchAll() as $row) {
            $settings[$row['setting_key']] = $row['setting_value'];
        }
    }
} catch (\Throwable $e) {
    $settings = [];
}

$defaultSettings = [
    'hero_heading_text'    => 'Welcome to St. Patrick\'s Parish Office',
    'hero_eyebrow_text'    => 'Parish Office Online',
    'hero_subtitle_text'   => 'Dingalan, Aurora - the parish\'s sacrament records and certificates, kept faithfully for generations, now easier to reach.',
    'office_hours_text'    => "Tuesday - Friday: 8:00 AM - 5:00 PM\nSaturday: 8:00 AM - 12:00 NN\nSunday & Monday: Closed",
    'office_location_text' => "St. Patrick's Parish Office\nDingalan, Aurora, Philippines",
];
foreach ($defaultSettings as $key => $value) {
    if (!isset($settings[$key])) {
        $settings[$key] = $value;
    }
}


$pageTitle = 'Home';

// Convert PHP parse-time constants (timezone rows, after-today flag) into
// per-event metadata that the FullCalendar plugin reads.
//
// - Color is driven by schedule_events.event_type + the map below. Events
//   whose type the map does not recognize fall back to a neutral grey so the
//   calendar stays readable even for unexpected types.
//
// - Events are large enough to read because:
//   * dayMaxEventRows + eventDisplay: 'block' let each event grow to its full
//     content instead of clipping to "+N more".
//   * displayEventTime: true shows the time in 12-hour format.
//   * fixedWeekCount: false drops the empty trailing week.
//   * text overflow / truncation is disabled so long titles wrap normally.
$typeColorMap = [
    'mass'       => '#2a7fe0',   // weekdays / holy days reference blue
    'wedding'    => '#d14b8b',   // celebrations - warm pink
    'confirmation' => '#8a5fc9',
    'baptism'    => '#2a9d8f',   // baptism color - teal
    'feast'      => '#e8a049',   // feast days - warm gold
    'activity'   => '#7c6bc0',   // other parish activities - soft violet
    'lecture'    => '#37a694',
];
function schedule_event_color($type, $map) {
    return isset($map[$type]) ? $map[$type] : '#7a8797';
}
